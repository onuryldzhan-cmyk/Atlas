from __future__ import annotations

logger = logging.getLogger(
    "AtlasAI.MarketLoop"

class MarketLoop:


    VERSION = "7.0"



    def __init__(

        self,

        brain,

        market_feed,

        execution=None,

        event_bus=None,

        symbols=None,

        interval=10

    ):


        self.brain = brain

        self.market_feed = market_feed

        self.execution = execution

        self.event_bus = event_bus



        self.symbols = symbols or [

            "BTCUSDT",
            "ETHUSDT",
            "BNBUSDT",
            "SOLUSDT",
            "XRPUSDT"

        ]



        self.interval = interval



        self.processor = CandleProcessor()

        self.technical = TechnicalAnalyzer()

        self.world = WorldAnalysisEngine()



        self.running = False

        self.stop_event = threading.Event()

        self.thread = None



        self.started_at = None

        self.last_cycle = None



        self.last_signals = {}



        self.metrics = {

            "cycles":0,

            "symbols_scanned":0,

            "decisions":0,

            "executions":0,

            "errors":0,

            "latency":0

        }



        self.errors = []



        logger.info(

            "Atlas Market Loop v%s initialized",

            self.VERSION

        )






    # =====================================================
    # START
    # =====================================================


    def start(self):


        if self.running:

            return



        self.running = True

        self.stop_event.clear()



        self.started_at = datetime.now(
            UTC
        ).isoformat()



        self.thread = threading.Thread(

            target=self.worker,

            name="AtlasMarketLoop",

            daemon=True

        )



        self.thread.start()



        logger.info(
            "Market Loop started"
        )






    # =====================================================
    # STOP
    # =====================================================


    def stop(self):


        self.running = False


        self.stop_event.set()



        if self.thread:


            self.thread.join(
                timeout=5
            )



        logger.info(
            "Market Loop stopped"
        )






    # =====================================================
    # WORKER
    # =====================================================


    def worker(self):


        while self.running:


            start=time.time()


            try:


                for symbol in self.symbols:


                    if not self.running:

                        break


                    self.process_symbol(
                        symbol
                    )



                self.metrics["cycles"] += 1



                self.last_cycle = datetime.now(
                    UTC
                ).isoformat()



                self.metrics["latency"] = round(

                    time.time()-start,

                    4

                )



            except Exception as e:


                self.metrics["errors"] += 1


                self.errors.append({

                    "time":
                    datetime.now(
                        UTC
                    ).isoformat(),

                    "error":
                    str(e)

                })


                logger.exception(
                    "Market cycle failed"
                )



            self.stop_event.wait(
                self.interval
            )








    # =====================================================
    # SYMBOL PIPELINE
    # =====================================================


    def process_symbol(

        self,

        symbol

    ):


        try:


            candles = self.market_feed.candles(

                symbol,

                interval="1m",

                limit=200

            )


            if not candles:

                return



            candles = self.processor.normalize(
                candles
            )



            if len(candles)<50:

                return




            analysis = self.technical.analyze(

                candles

            )



            price = self.processor.latest_price(

                candles

            )


            if not price:

                return



            score = self.calculate_score(

                analysis

            )



            market_data = self.world.build_market_data(

                symbol,

                price,

                analysis,

                score

            )



            decision = self.brain.analyze_and_decide(

                symbol,

                market_data

            )



            if not decision:

                return



            decision["symbol"] = symbol

            decision["price"] = price



            self.last_signals[symbol] = decision



            self.metrics["decisions"] += 1



            self.publish(

                "ATLAS_DECISION",

                decision

            )



            state = self.world.build_market_state(

                market_data

            )



            if self.execution:


                result = self.execution.execute(

                    decision,

                    state

                )


                self.metrics["executions"] += 1



                logger.info(

                    "%s execution %s",

                    symbol,

                    result.get(
                        "status"
                    )

                )



            self.metrics["symbols_scanned"] +=1




        except Exception as e:


            self.metrics["errors"] +=1


            logger.exception(

                "%s processing failed",

                symbol

            )







    # =====================================================
    # SCORE
    # =====================================================


    def calculate_score(

        self,

        analysis

    ):


        score=0.5



        rsi = analysis.get(
            "rsi"
        )


        trend = analysis.get(
            "trend"
        )



        if rsi:


            if 40 <= rsi <=60:

                score +=0.15


            elif rsi <30:

                score +=0.10


            elif rsi >70:

                score -=0.10



        if trend=="BULLISH":

            score+=0.20



        elif trend=="BEARISH":

            score-=0.20



        return round(

            max(

                0,

                min(

                    score,

                    1

                )

            ),

            2

        )

    # EVENT BUS
    # =====================================================


    def publish(

        event,

        data

    ):


        if not self.event_bus:

                    data

                )


        except Exception:


            logger.exception(

                "Event publish failed"

            )

            "running":

            self.running,


            "started_at":

            self.started_at,


            "symbols":

            len(self.symbols),


            "metrics":

            self.metrics,


            "last_cycle":

            self.last_cycle,


            "errors":

            len(self.errors)

        }

    # ============================================================


    def execute(

        self,

        decision,

        market_state=None

    ):


        self.stats["received"] += 1



        if not self.running:


            return self.reject(

                "EXECUTION_MANAGER_OFFLINE"

            )



        try:



            request = self.build_request(

                decision

            )



            validation = self.validate_request(

                request

            )



            if not validation["approved"]:


                return self.reject(

                    validation["reason"]

                )




            risk_check = self.risk_check(

                request,

                market_state

            )



            if not risk_check["approved"]:


                return self.reject(

                    risk_check["reason"]

                )





            if self.mode == "PAPER":



                result = self.paper_execute(

                    request

                )


            else:



                result = self.live_execute(

                    request

                )





            self.stats["executed"] += 1



            self.history.append(

                result

            )



            self.publish(

                "EXECUTION_COMPLETED",

                result

            )



            return result





        except Exception as e:



            self.stats["failed"] += 1



            logger.exception(

                "Execution pipeline failed"

            )



            return {


                "status":

                "FAILED",


                "error":

                str(e)

            }

from __future__ import annotations

class DecisionAdapter:



    VERSION = "4.0"



    def __init__(self):


        self.converted = 0

        self.errors = []







    # =====================================================
    # PUBLIC ENTRY
    # =====================================================


    @classmethod
    def convert(

        cls,

        decision

    ):


        adapter = cls()


        return adapter.normalize(

            decision

        )








    # =====================================================
    # NORMALIZER
    # =====================================================


    def normalize(

        self,

        decision

    ):



        try:



            if decision is None:


                return None





            data = self.to_dict(

                decision

            )



            if data is None:


                return None





            result = self.build_schema(

                data

            )



            self.converted += 1



            return result





        except Exception as e:



            logger.exception(

                "Decision normalization failed"

            )



            self.errors.append(

                str(e)

            )


            return None








    # =====================================================
    # OBJECT CONVERSION
    # =====================================================


    def to_dict(

        self,

        decision

    ):



        if isinstance(

            decision,

            dict

        ):


            return decision





        if is_dataclass(

            decision

        ):


            return asdict(

                decision

            )





        if hasattr(

            decision,

            "__dict__"

        ):


            return vars(

                decision

            )





        logger.error(

            "Unsupported decision type: %s",

            type(decision)

        )



        return None








    # =====================================================
    # BUILD ATLAS FORMAT
    # =====================================================


    def build_schema(

        self,

        data

    ):



        action = self.get_action(

            data

        )


        symbol = self.get_symbol(

            data

        )


        confidence = self.get_confidence(

            data

        )


        price = self.get_price(

            data

        )


        return {


            "action":

            action,



            "symbol":

            symbol,



            "confidence":

            confidence,



            "strategy":

            data.get(

                "strategy",

                "Atlas"

            ),



            "risk_level":

            str(

                data.get(

                    "risk_level",

                    data.get(

                        "risk",

                        "MEDIUM"

                    )

                )

            ).upper(),



            "price":

            price,



            "created_at":

            datetime.now(

                UTC

            ).isoformat(),



            "metadata":

            {


                "adapter_version":

                self.VERSION,


                "source":

                "DecisionAdapter"

            }

        }










    # =====================================================
    # FIELD EXTRACTION
    # =====================================================


    def get_action(

        self,

        data

    ):



        action = data.get(

            "action",

            data.get(

                "decision",

                "HOLD"

            )

        )



        action = str(

            action

        ).upper()





        aliases = {


            "LONG":

            "BUY",


            "SHORT":

            "SELL",

from __future__ import annotations

from abc import ABC, abstractmethod
from typing import Protocol

from core.execution_models import (
    ExecutionRequest,
    ExecutionResult,
)


# ==========================================================
# MAIN EXECUTION CONTRACT
# ==========================================================

class ExecutionEngine(ABC):

    @property
    @abstractmethod
    def name(self) -> str:
        """Engine name"""

    @property
    @abstractmethod
    def mode(self) -> str:
        """Execution mode"""

    @abstractmethod
    def execute(
        self,
        request: ExecutionRequest,
    ) -> ExecutionResult:
        """
        Execute one order.
        """

    @abstractmethod
    def health(self) -> dict:
        """
        Component health information.
        """

    @abstractmethod
    def status(self) -> dict:
        """
        Runtime statistics.
        """

    @abstractmethod
    def reset(self) -> None:
        """
        Reset runtime state.
        """

# OPTIONAL PROTOCOLS

    def health(self) -> dict:
        ...


class SupportsStatus(Protocol):

    def status(self) -> dict:
        ...


class SupportsReset(Protocol):

    def reset(self) -> None:
        ...


class SupportsDiagnostics(Protocol):

    def diagnostics(self) -> dict:
        ...
