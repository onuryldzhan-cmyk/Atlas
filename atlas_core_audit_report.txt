  255: class AtlasProcess:
  256:     """
  257:     Controls the Atlas process lifecycle.
  258: 
  259:     Signal handlers do not perform shutdown work directly. They only set
  260:     the shutdown event so the main thread can stop Atlas safely.
  261:     """
  262: 
  263:     VERSION = "1.0"
  264: 
  265:     def __init__(self) -> None:
  266:         self.application: AtlasApplication | None = None
  267: 
  268:         self.shutdown_event = threading.Event()
  269: 
  270:         self.shutdown_reason = ""
  271: 
  272:         self.started_at = datetime.now(
  273:             UTC
  274:         ).isoformat()
  275: 
  276:         self.exit_code = 0
  277: 
  278:     # --------------------------------------------------------
  279:     # SIGNALS
  280:     # --------------------------------------------------------
  281: 
  282:     def install_signal_handlers(self) -> None:
  283:         """
  284:         Register operating-system termination handlers.
  285:         """
  286: 
  287:         signal.signal(
  288:             signal.SIGINT,
  289:             self._handle_signal,
  290:         )
  291: 
  292:         if hasattr(signal, "SIGTERM"):
  293:             signal.signal(
  294:                 signal.SIGTERM,
  295:                 self._handle_signal,
  296:             )
  297: 
  298:     def _handle_signal(
  299:         self,
  300:         signal_number: int,
  301:         frame: Any,
  302:     ) -> None:
  303:         """
  304:         Request a graceful shutdown.
  305:         """
  306: 
  307:         del frame
  308: 
  309:         if self.shutdown_event.is_set():
  310:             return
  311: 
  312:         try:
  313:             signal_name = signal.Signals(
  314:                 signal_number
  315:             ).name
  316:         except ValueError:
  317:             signal_name = str(
  318:                 signal_number
  319:             )
  320: 
  321:         self.shutdown_reason = (
  322:             f"Operating-system signal: {signal_name}"
  323:         )
  324: 
  325:         logger.warning(
  326:             "Shutdown requested: %s",
  327:             self.shutdown_reason,
  328:         )
  329: 
  330:         self.shutdown_event.set()
  331: 
  332:     # --------------------------------------------------------
  333:     # START
  334:     # --------------------------------------------------------
  335: 
  336:     def start(self) -> bool:
  337:         """
  338:         Construct and start Atlas.
  339:         """
  340: 
  341:         logger.info(
  342:             "Constructing AtlasApplication"
  343:         )
  344: 
  345:         self.application = AtlasApplication(
  346:             execution_mode="PAPER",
  347:         )
  348: 
  349:         logger.info(
  350:             "Starting AtlasApplication"
  351:         )
  352: 
  353:         started = self.application.start()
  354: 
  355:         if not started:
  356:             self.exit_code = 1
  357: 
  358:             logger.error(
  359:                 "AtlasApplication reported startup failure"
  360:             )
  361: 
  362:             self._display_latest_error()
  363: 
  364:             return False
  365: 
  366:         display_startup_status(
  367:             self.application
  368:         )
  369: 
  370:         return True
  371: 
  372:     # --------------------------------------------------------
  373:     # MAIN LOOP
  374:     # --------------------------------------------------------
  375: 
  376:     def run_forever(self) -> None:
  377:         """
  378:         Keep the primary process alive until shutdown is requested.
  379: 
  380:         Market analysis remains the responsibility of MarketLoop.
  381:         This loop only supervises the application process.
  382:         """
  383: 
  384:         logger.info(
  385:             "Atlas main process is active"
  386:         )
  387: 
  388:         logger.info(
  389:             "Press Ctrl+C to stop Atlas safely"
  390:         )
  391: 
  392:         while not self.shutdown_event.wait(
  393:             timeout=1.0
  394:         ):
  395:             if self.application is None:
  396:                 self.exit_code = 1
  397:                 self.shutdown_reason = (
  398:                     "Application instance disappeared"
  399:                 )
  400:                 break
  401: 
  402:             if not self.application.running:
  403:                 self.exit_code = 1
  404:                 self.shutdown_reason = (
  405:                     "AtlasApplication stopped unexpectedly"
  406:                 )
  407: 
  408:                 logger.error(
  409:                     self.shutdown_reason
  410:                 )
  411: 
  412:                 break

  414:     # --------------------------------------------------------
  415:     # ERROR DETAILS
  416:     # --------------------------------------------------------
  417: 
  418:     def _display_latest_error(self) -> None:
  419:         """
  420:         Log the latest structured application error when available.
  421:         """
  422: 
  423:         if self.application is None:
  424:             return
  425: 
  426:         status = safe_dict(
  427:             self.application.status()
  428:         )
  429: 
  430:         errors = safe_dict(
  431:             status.get("errors")
  432:         )
  433: 
  434:         latest = errors.get(
  435:             "latest"
  436:         )
  437: 
  438:         if not isinstance(latest, dict):
  439:             return
  440: 
  441:         logger.error(
  442:             "Application error phase: %s",
  443:             latest.get(
  444:                 "phase",
  445:                 "UNKNOWN",
  446:             ),
  447:         )
  448: 
  449:         logger.error(
  450:             "Application error type: %s",
  451:             latest.get(
  452:                 "type",
  453:                 "UNKNOWN",
  454:             ),
  455:         )
  456: 
  457:         logger.error(
  458:             "Application error: %s",
  459:             latest.get(
  460:                 "error",
  461:                 "UNKNOWN",
  462:             ),
  463:         )
  464: 
  465:     # --------------------------------------------------------
  466:     # STOP
  467:     # --------------------------------------------------------
  468: 
  469:     def stop(self) -> bool:
  470:         """
  471:         Stop Atlas safely.
  472:         """
  473: 
  474:         if self.application is None:
  475:             return True
  476: 
  477:         logger.info(
  478:             "Stopping AtlasApplication"
  479:         )
  480: 
  481:         stopped = self.application.stop()
  482: 
  483:         if not stopped:
  484:             self.exit_code = max(
  485:                 self.exit_code,
  486:                 1,
  487:             )
  488: 
  489:             logger.warning(
  490:                 "Atlas stopped with one or more shutdown errors"
  491:             )
  492: 
  493:         return stopped
  494: 
  495:     # --------------------------------------------------------
  496:     # COMPLETE PROCESS
  497:     # --------------------------------------------------------
  498: 
  499:     def execute(self) -> int:
  500:         """
  501:         Run the full Atlas process lifecycle.
  502:         """
  503: 
  504:         self.install_signal_handlers()
  505: 
  506:         try:
  507:             if not self.start():
  508:                 return self.exit_code
  509: 
  510:             self.run_forever()
  511: 
  512:         except KeyboardInterrupt:
  513:             self.shutdown_reason = (
  514:                 "Keyboard interrupt"
  515:             )
  516: 
  517:             logger.warning(
  518:                 "Keyboard interrupt received"
  519:             )
  520: 
  521:         except Exception as exc:
  522:             self.exit_code = 1
  523: 
  524:             self.shutdown_reason = (
  525:                 f"Fatal error: {exc}"
  526:             )
  527: 
  528:             logger.critical(
  529:                 "Fatal Atlas error: %s",
  530:                 exc,
  531:             )
  532: 
  533:             logger.debug(
  534:                 "Fatal traceback:\n%s",
  535:                 traceback.format_exc(),
  536:             )
  537: 
  538:         finally:
  539:             try:
  540:                 self.stop()
  541:             except Exception:
  542:                 self.exit_code = 1
  543: 
  544:                 logger.exception(
  545:                     "Fatal error during Atlas shutdown"
  546:                 )
  547: 
  548:             logger.info(
  549:                 "Atlas process finished | "
  550:                 "exit_code=%s | reason=%s",
  551:                 self.exit_code,
  552:                 (
  553:                     self.shutdown_reason
  554:                     or "Normal shutdown"
  555:                 ),
  556:             )
  557: 
  558:         return self.exit_code
  559: 
  560:

  562: # PUBLIC MAIN
  563: # ============================================================
  564:

  567:     """

  570:

  572: 
  573:     print()
  574:     print("=" * 64)
  575:     print("PROJECT ATLAS AI")
  576:     print("Institutional AI Investment Platform")
  577:     print("=" * 64)
  578:     print()
  579: 
  580:     process = AtlasProcess()
  581: 
  582:     return process.execute()

  249: class AtlasRuntime:
  250:     """
  251:     Production-oriented process lifecycle manager for Atlas AI.
  252: 
  253:     AtlasRuntime is intentionally small. It does not implement
  254:     trading, analysis, decision-making, execution, learning, or
  255:     research logic.
  256: 
  257:     Its responsibility is to assemble and control AtlasOS services.
  258:     """
  259: 
  260:     VERSION = "1.0.0"
  261: 
  262:     NAME = "AtlasRuntime"
  263: 
  264:     def __init__(
  265:         self,
  266:         *,

  268:         metadata: Mapping[
  269:             str,
  270:             Any,
  271:         ] | None = None,
  272:         install_signal_handlers: bool = True,
  273:     ) -> None:
  274:         """
  275:         Initialize Atlas Runtime.
  276: 
  277:         Parameters
  278:         ----------
  279:         kernel:
  280:             Optional existing AtlasOSKernel. A new kernel is created
  281:             when omitted.
  282: 
  283:         metadata:
  284:             Plaintext-safe runtime metadata.
  285: 
  286:         install_signal_handlers:
  287:             Whether SIGINT and SIGTERM handlers should be installed
  288:             during startup.
  289:         """
  290: 
  291:         if (
  292:             kernel is not None
  293:             and not isinstance(
  294:                 kernel,
  295:                 AtlasOSKernel,
  296:             )
  297:         ):
  298:             raise TypeError(
  299:                 "kernel must be an AtlasOSKernel "
  300:                 "instance or None."
  301:             )
  302: 
  303:         self._kernel = (
  304:             kernel
  305:             if kernel is not None
  306:             else AtlasOSKernel(
  307:                 metadata={
  308:                     "runtime": self.NAME,
  309:                     "runtime_version": self.VERSION,
  310:                 }
  311:             )
  312:         )
  313: 
  314:         self._metadata: dict[
  315:             str,
  316:             Any,
  317:         ] = dict(
  318:             metadata or {}
  319:         )
  320: 
  321:         self._install_signal_handlers = bool(
  322:             install_signal_handlers
  323:         )
  324: 
  325:         self._state = (
  326:             AtlasRuntimeState.CREATED
  327:         )
  328: 
  329:         self._created_at = datetime.now(
  330:             UTC
  331:         )
  332: 
  333:         self._started_at: datetime | None = None
  334: 
  335:         self._stopped_at: datetime | None = None
  336: 
  337:         self._last_error: str | None = None
  338: 
  339:         self._registrations: dict[
  340:             str,
  341:             RuntimeServiceRegistration,
  342:         ] = {}
  343:

  345:             str

  348:         self._shutdown_event = (
  349:             threading.Event()
  350:         )
  351:

  353: 
  354:         self._signal_handlers_installed = False
  355: 
  356:         self._previous_signal_handlers: dict[
  357:             signal.Signals,
  358:             Any,
  359:         ] = {}
  360: 
  361:         logger.info(
  362:             "%s v%s initialized.",
  363:             self.NAME,
  364:             self.VERSION,
  365:         )

  504:     def register_service(
  505:         self,
  506:         service: AtlasService,
  507:         *,
  508:         dependencies: Iterable[
  509:             str
  510:         ] = (),
  511:         critical: bool = True,
  512:     ) -> RuntimeServiceRegistration:
  513:         """
  514:         Register one AtlasOS service and its dependencies.
  515: 
  516:         Services may only be registered before Runtime startup.
  517:         """
  518: 
  519:         if service is None:
  520:             raise TypeError(
  521:                 "service is required."
  522:             )
  523: 
  524:         with self._lock:
  525:             self._ensure_state(
  526:                 allowed={
  527:                     AtlasRuntimeState.CREATED,
  528:                     AtlasRuntimeState.CONFIGURING,
  529:                     AtlasRuntimeState.READY,
  530:                     AtlasRuntimeState.STOPPED,
  531:                 },
  532:                 operation="register service",
  533:             )
  534: 
  535:             self._state = (
  536:                 AtlasRuntimeState.CONFIGURING
  537:             )
  538: 
  539:             service_name = self._service_name(
  540:                 service
  541:             )
  542: 
  543:             if service_name in self._registrations:
  544:                 raise AtlasRuntimeConfigurationError(
  545:                     "Service is already registered with "
  546:                     f"Runtime: {service_name}"
  547:                 )
  548: 
  549:             normalized_dependencies = (
  550:                 self._normalize_dependencies(
  551:                     dependencies
  552:                 )
  553:             )
  554: 
  555:             self._kernel.register_service(
  556:                 service,
  557:                 dependencies=(
  558:                     normalized_dependencies
  559:                 ),
  560:             )
  561: 
  562:             registration = (
  563:                 RuntimeServiceRegistration(
  564:                     service_name=service_name,
  565:                     dependencies=(
  566:                         normalized_dependencies
  567:                     ),
  568:                     critical=bool(
  569:                         critical
  570:                     ),
  571:                 )
  572:             )
  573: 
  574:             self._registrations[
  575:                 service_name
  576:             ] = registration
  577: 
  578:             self._registration_order.append(
  579:                 service_name
  580:             )
  581: 
  582:             logger.info(
  583:                 "Runtime service registered: %s; "
  584:                 "dependencies=%s; critical=%s",
  585:                 service_name,
  586:                 normalized_dependencies,
  587:                 critical,
  588:             )
  589: 
  590:             return registration
  591: 
  592:     def register_services(
  593:         self,
  594:         registrations: Iterable[
  595:             tuple[
  596:                 AtlasService,
  597:                 Iterable[str],
  598:             ]
  599:         ],
  600:     ) -> tuple[
  601:         RuntimeServiceRegistration,
  602:         ...,
  603:     ]:
  604:         """
  605:         Register multiple services.
  606: 
  607:         Each item must contain:
  608: 
  609:             (service, dependencies)
  610:         """
  611: 
  612:         completed: list[
  613:             RuntimeServiceRegistration
  614:         ] = []
  615: 
  616:         for service, dependencies in registrations:
  617:             completed.append(
  618:                 self.register_service(
  619:                     service,
  620:                     dependencies=dependencies,
  621:                 )
  622:             )
  623: 
  624:         return tuple(
  625:             completed
  626:         )

  628:     def validate(
  629:         self,
  630:     ) -> None:
  631:         """
  632:         Validate Runtime and Kernel configuration.
  633:         """
  634: 
  635:         with self._lock:
  636:             self._ensure_state(
  637:                 allowed={
  638:                     AtlasRuntimeState.CREATED,
  639:                     AtlasRuntimeState.CONFIGURING,
  640:                     AtlasRuntimeState.READY,
  641:                     AtlasRuntimeState.STOPPED,
  642:                 },
  643:                 operation="validate runtime",
  644:             )
  645: 
  646:             try:
  647:                 self._kernel.validate()
  648: 
  649:             except Exception as error:
  650:                 self._state = (
  651:                     AtlasRuntimeState.FAILED
  652:                 )
  653: 
  654:                 self._last_error = (
  655:                     f"{type(error).__name__}: "
  656:                     f"{error}"
  657:                 )
  658: 
  659:                 raise AtlasRuntimeConfigurationError(
  660:                     "Atlas Runtime configuration "
  661:                     "validation failed."
  662:                 ) from error
  663: 
  664:             self._state = (
  665:                 AtlasRuntimeState.READY
  666:             )
  667: 
  668:             self._last_error = None
  669: 
  670:             logger.info(
  671:                 "Atlas Runtime validation passed. "
  672:                 "Registered services: %s.",
  673:                 self.service_count,
  674:             )

  677:     # STARTUP
  678:     # =====================================================
  679: 
  680:     def start(
  681:         self,
  682:         *,
  683:         rollback_on_failure: bool = True,
  684:     ) -> Any:
  685:         """
  686:         Start Atlas Runtime and all registered services.
  687: 
  688:         The returned value is the AtlasOS Kernel orchestration
  689:         result.
  690:         """
  691: 
  692:         with self._lock:
  693:             if (
  694:                 self._state
  695:                 is AtlasRuntimeState.RUNNING
  696:             ):
  697:                 logger.info(
  698:                     "Atlas Runtime is already running."
  699:                 )
  700: 
  701:                 return None
  702: 
  703:             self._ensure_state(
  704:                 allowed={
  705:                     AtlasRuntimeState.CREATED,
  706:                     AtlasRuntimeState.CONFIGURING,
  707:                     AtlasRuntimeState.READY,
  708:                     AtlasRuntimeState.STOPPED,
  709:                     AtlasRuntimeState.FAILED,
  710:                 },
  711:                 operation="start runtime",
  712:             )
  713: 
  714:             self._state = (
  715:                 AtlasRuntimeState.STARTING
  716:             )
  717: 
  718:             self._last_error = None
  719: 
  720:             self._stopped_at = None
  721: 
  722:             self._shutdown_event.clear()
  723: 
  724:         logger.info(
  725:             "Atlas Runtime startup initiated."
  726:         )
  727: 
  728:         try:
  729:             self._kernel.validate()
  730: 
  731:             if self._install_signal_handlers:
  732:                 self._install_shutdown_handlers()
  733: 
  734:             result = self._kernel.start(
  735:                 rollback_on_failure=(
  736:                     rollback_on_failure
  737:                 )
  738:             )
  739: 
  740:             if not self._kernel.is_running:
  741:                 raise AtlasRuntimeStartupError(
  742:                     "AtlasOS Kernel did not enter "
  743:                     "RUNNING state."
  744:                 )
  745: 
  746:             with self._lock:
  747:                 self._started_at = datetime.now(
  748:                     UTC
  749:                 )
  750: 
  751:                 self._state = (
  752:                     AtlasRuntimeState.RUNNING
  753:                 )
  754: 
  755:             logger.info(
  756:                 "Atlas Runtime started successfully. "
  757:                 "Services: %s.",
  758:                 self.service_count,
  759:             )
  760: 
  761:             return result
  762: 
  763:         except Exception as error:
  764:             with self._lock:
  765:                 self._state = (
  766:                     AtlasRuntimeState.FAILED
  767:                 )
  768: 
  769:                 self._last_error = (
  770:                     f"{type(error).__name__}: "
  771:                     f"{error}"
  772:                 )
  773: 
  774:             logger.exception(
  775:                 "Atlas Runtime startup failed."
  776:             )
  777: 
  778:             if self._install_signal_handlers:
  779:                 self._restore_signal_handlers()
  780: 
  781:             if isinstance(
  782:                 error,
  783:                 AtlasRuntimeError,
  784:             ):
  785:                 raise
  786: 
  787:             raise AtlasRuntimeStartupError(
  788:                 "Atlas Runtime startup failed."
  789:             ) from error
  790:

  795:     def request_shutdown(
  796:         self,
  797:         *,
  798:         reason: str = "Shutdown requested.",
  799:     ) -> None:
  800:         """
  801:         Request Runtime shutdown.
  802: 
  803:         This method only sets the shutdown event. It does not perform
  804:         shutdown immediately, making it safe for signal handlers.
  805:         """
  806: 
  807:         normalized_reason = (
  808:             str(
  809:                 reason
  810:             ).strip()
  811:             or "Shutdown requested."
  812:         )
  813: 
  814:         logger.warning(
  815:             "Atlas Runtime shutdown requested: %s",
  816:             normalized_reason,
  817:         )
  818: 
  819:         self._shutdown_event.set()
  820: 
  821:     def stop(
  822:         self,
  823:         *,
  824:         suppress_errors: bool = False,
  825:     ) -> Any:
  826:         """
  827:         Stop Atlas Runtime and AtlasOS Kernel safely.
  828:         """
  829: 
  830:         with self._lock:
  831:             if self._state in {
  832:                 AtlasRuntimeState.CREATED,
  833:                 AtlasRuntimeState.READY,
  834:                 AtlasRuntimeState.STOPPED,
  835:             }:
  836:                 self._state = (
  837:                     AtlasRuntimeState.STOPPED
  838:                 )
  839: 
  840:                 self._stopped_at = datetime.now(
  841:                     UTC
  842:                 )
  843: 
  844:                 self._shutdown_event.set()
  845: 
  846:                 self._restore_signal_handlers()
  847: 
  848:                 logger.info(
  849:                     "Atlas Runtime stopped."
  850:                 )
  851: 
  852:                 return None
  853: 
  854:             if (
  855:                 self._state
  856:                 is AtlasRuntimeState.STOPPING
  857:             ):
  858:                 return None
  859: 
  860:             self._state = (
  861:                 AtlasRuntimeState.STOPPING
  862:             )
  863: 
  864:         logger.info(
  865:             "Atlas Runtime shutdown initiated."
  866:         )
  867: 
  868:         try:
  869:             result = self._kernel.stop()
  870: 
  871:             with self._lock:
  872:                 self._state = (
  873:                     AtlasRuntimeState.STOPPED
  874:                 )
  875: 
  876:                 self._stopped_at = datetime.now(
  877:                     UTC
  878:                 )
  879: 
  880:                 self._shutdown_event.set()
  881: 
  882:                 self._last_error = None
  883: 
  884:             logger.info(
  885:                 "Atlas Runtime stopped successfully."
  886:             )
  887: 
  888:             return result
  889: 
  890:         except Exception as error:
  891:             with self._lock:
  892:                 self._state = (
  893:                     AtlasRuntimeState.FAILED
  894:                 )
  895: 
  896:                 self._last_error = (
  897:                     f"{type(error).__name__}: "
  898:                     f"{error}"
  899:                 )
  900: 
  901:                 self._stopped_at = datetime.now(
  902:                     UTC
  903:                 )
  904: 
  905:                 self._shutdown_event.set()
  906: 
  907:             logger.exception(
  908:                 "Atlas Runtime shutdown failed."
  909:             )
  910: 
  911:             if suppress_errors:
  912:                 return None
  913: 
  914:             raise AtlasRuntimeShutdownError(
  915:                 "Atlas Runtime shutdown failed."
  916:             ) from error
  917: 
  918:         finally:
  919:             self._restore_signal_handlers()
  920: 
  921:     def run_forever(
  922:         self,
  923:         *,
  924:         poll_interval: float = 0.5,
  925:     ) -> None:
  926:         """
  927:         Start Runtime and block until shutdown is requested.
  928:         """
  929: 
  930:         interval = float(
  931:             poll_interval
  932:         )
  933: 
  934:         if interval <= 0:
  935:             raise ValueError(
  936:                 "poll_interval must be greater than zero."
  937:             )
  938: 
  939:         self.start()
  940: 
  941:         try:
  942:             while not self.shutdown_requested:
  943:                 time.sleep(
  944:                     interval
  945:                 )
  946: 
  947:         except KeyboardInterrupt:
  948:             self.request_shutdown(
  949:                 reason="Keyboard interrupt."
  950:             )
  951: 
  952:         finally:
  953:             self.stop(
  954:                 suppress_errors=False
  955:             )

  490: 
  491:     def start(
  492:         self,
  493:         *,
  494:         rollback_on_failure: bool = True,
  495:     ) -> OrchestrationResult:
  496:         """
  497:         Start AtlasOS and all registered services.
  498: 
  499:         Startup is idempotent when the kernel is already running.
  500:         """
  501: 
  502:         with self._lock:
  503: 
  504:             if self._state in {
  505:                 KernelState.RUNNING,
  506:                 KernelState.DEGRADED,
  507:             }:
  508: 
  509:                 if self._last_start_result is not None:
  510:                     return self._last_start_result
  511: 
  512:                 return OrchestrationResult(
  513:                     operation="START",
  514:                     success=True,
  515:                     steps=(),
  516:                 )
  517: 
  518:             if self._state in {
  519:                 KernelState.STARTING,
  520:                 KernelState.STOPPING,
  521:             }:
  522:                 raise KernelLifecycleError(
  523:                     f"Kernel cannot start while state is "
  524:                     f"{self._state.value}."
  525:                 )
  526: 
  527:             self._state = KernelState.STARTING
  528:             self._last_error = None
  529: 
  530:             try:
  531: 
  532:                 self.validate()
  533: 
  534:                 self._startup_plan = self.plan()
  535: 
  536:                 result = self._service_manager.start_all(
  537:                     rollback_on_failure=rollback_on_failure,
  538:                 )
  539: 
  540:                 self._last_start_result = result
  541: 
  542:                 if result.success:
  543: 
  544:                     self._started_at = datetime.now(UTC)
  545:                     self._stopped_at = None
  546: 
  547:                     service_health = (
  548:                         self.registry.health()
  549:                     )
  550: 
  551:                     if (
  552:                         service_health.status
  553:                         == HealthStatus.DEGRADED
  554:                     ):
  555:                         self._state = (
  556:                             KernelState.DEGRADED
  557:                         )
  558:                     else:
  559:                         self._state = (
  560:                             KernelState.RUNNING
  561:                         )
  562: 
  563:                     logger.info(
  564:                         "AtlasOS kernel started: %s v%s | "
  565:                         "services=%s",
  566:                         self.identity.name,
  567:                         self.identity.version,
  568:                         self.service_count,
  569:                     )
  570: 
  571:                 else:
  572: 
  573:                     self._state = KernelState.FAILED
  574: 
  575:                     self._last_error = (
  576:                         "One or more AtlasOS services failed "
  577:                         "during startup."
  578:                     )
  579: 
  580:                     logger.error(
  581:                         "AtlasOS kernel startup failed."
  582:                     )
  583: 
  584:                 return result
  585: 
  586:             except Exception as exc:
  587: 
  588:                 self._last_error = (
  589:                     f"{type(exc).__name__}: {exc}"
  590:                 )
  591: 
  592:                 self._state = KernelState.FAILED
  593: 
  594:                 logger.exception(
  595:                     "AtlasOS kernel lifecycle startup failure."
  596:                 )
  597: 
  598:                 raise
  599:

  604:     def stop(self) -> OrchestrationResult:
  605:         """
  606:         Stop all AtlasOS services safely.
  607: 
  608:         Shutdown is idempotent.
  609:         """
  610: 
  611:         with self._lock:
  612: 
  613:             if self._state in {
  614:                 KernelState.CREATED,
  615:                 KernelState.STOPPED,
  616:             }:
  617: 
  618:                 self._state = KernelState.STOPPED
  619: 
  620:                 if self._stopped_at is None:
  621:                     self._stopped_at = datetime.now(UTC)
  622: 
  623:                 if self._last_stop_result is not None:
  624:                     return self._last_stop_result
  625: 
  626:                 result = OrchestrationResult(
  627:                     operation="STOP",
  628:                     success=True,
  629:                     steps=(),
  630:                 )
  631: 
  632:                 self._last_stop_result = result
  633: 
  634:                 return result
  635: 
  636:             if self._state == KernelState.STOPPING:
  637: 
  638:                 if self._last_stop_result is not None:
  639:                     return self._last_stop_result
  640: 
  641:                 return OrchestrationResult(
  642:                     operation="STOP",
  643:                     success=True,
  644:                     steps=(),
  645:                 )
  646: 
  647:             if self._state == KernelState.STARTING:
  648:                 raise KernelLifecycleError(
  649:                     "Kernel cannot stop while startup "
  650:                     "is in progress."
  651:                 )
  652: 
  653:             self._state = KernelState.STOPPING
  654: 
  655:             try:
  656: 
  657:                 result = self._service_manager.stop_all()
  658: 
  659:                 self._last_stop_result = result
  660: 
  661:                 self._stopped_at = datetime.now(UTC)
  662: 
  663:                 if result.success:
  664: 
  665:                     self._state = KernelState.STOPPED
  666: 
  667:                     logger.info(
  668:                         "AtlasOS kernel stopped: %s",
  669:                         self.identity.name,
  670:                     )
  671: 
  672:                 else:
  673: 
  674:                     self._state = KernelState.FAILED
  675: 
  676:                     self._last_error = (
  677:                         "One or more AtlasOS services failed "
  678:                         "during shutdown."
  679:                     )
  680: 
  681:                     logger.error(
  682:                         "AtlasOS kernel shutdown completed "
  683:                         "with failures."
  684:                     )
  685: 
  686:                 return result
  687: 
  688:             except Exception as exc:
  689: 
  690:                 self._last_error = (
  691:                     f"{type(exc).__name__}: {exc}"
  692:                 )
  693: 
  694:                 self._state = KernelState.FAILED
  695: 
  696:                 logger.exception(
  697:                     "AtlasOS kernel lifecycle shutdown failure."
  698:                 )
  699: 
  700:                 raise

  706:     def health(self) -> KernelHealthReport:
  707:         """
  708:         Return aggregated AtlasOS kernel health.
  709:         """
  710: 
  711:         with self._lock:
  712: 
  713:             service_health = self.registry.health()
  714: 
  715:             if self._state == KernelState.RUNNING:
  716: 
  717:                 if (
  718:                     service_health.status
  719:                     == HealthStatus.UNHEALTHY
  720:                 ):
  721:                     status = HealthStatus.UNHEALTHY
  722:                     message = (
  723:                         "Kernel is running but one or more "
  724:                         "services are unhealthy."
  725:                     )
  726: 
  727:                 elif (
  728:                     service_health.status
  729:                     == HealthStatus.DEGRADED
  730:                 ):
  731:                     status = HealthStatus.DEGRADED
  732:                     message = (
  733:                         "Kernel is operating with degraded "
  734:                         "services."
  735:                     )
  736: 
  737:                 else:
  738:                     status = HealthStatus.HEALTHY
  739:                     message = (
  740:                         "Kernel and all services are operating "
  741:                         "normally."
  742:                     )
  743: 
  744:             elif self._state == KernelState.DEGRADED:
  745: 
  746:                 status = HealthStatus.DEGRADED
  747:                 message = (
  748:                     "Kernel is operating in degraded mode."
  749:                 )
  750: 
  751:             elif self._state == KernelState.FAILED:
  752: 
  753:                 status = HealthStatus.UNHEALTHY
  754:                 message = (
  755:                     self._last_error
  756:                     or "Kernel lifecycle failed."
  757:                 )
  758: 
  759:             else:
  760: 
  761:                 status = HealthStatus.UNKNOWN
  762:                 message = (
  763:                     f"Kernel is not operational. "
  764:                     f"State={self._state.value}."
  765:                 )
  766: 
  767:             return KernelHealthReport(
  768:                 kernel_name=self.identity.name,
  769:                 kernel_state=self._state,
  770:                 status=status,
  771:                 service_health=service_health,
  772:                 message=message,
  773:                 uptime_seconds=self.uptime_seconds,
  774:                 last_error=self._last_error,
  775:             )

  778:     # STATE SYNCHRONIZATION
  779:     # =====================================================
  780: 
  781:     def synchronize_state(self) -> KernelState:
  782:         """
  783:         Synchronize kernel state with current service health.
  784: 
  785:         This can be called by a future health monitor.
  786:         """
  787: 
  788:         with self._lock:
  789: 
  790:             if self._state not in {
  791:                 KernelState.RUNNING,
  792:                 KernelState.DEGRADED,
  793:             }:
  794:                 return self._state
  795: 
  796:             service_health = self.registry.health()
  797: 
  798:             if (
  799:                 service_health.status
  800:                 == HealthStatus.UNHEALTHY
  801:             ):
  802:                 self._state = KernelState.DEGRADED
  803: 
  804:                 self._last_error = (
  805:                     "One or more AtlasOS services are unhealthy."
  806:                 )
  807: 
  808:             elif (
  809:                 service_health.status
  810:                 == HealthStatus.DEGRADED
  811:             ):
  812:                 self._state = KernelState.DEGRADED
  813: 
  814:                 self._last_error = (
  815:                     "One or more AtlasOS services are degraded."
  816:                 )
  817: 
  818:             elif (
  819:                 service_health.status
  820:                 == HealthStatus.HEALTHY
  821:             ):
  822:                 self._state = KernelState.RUNNING
  823:                 self._last_error = None
  824: 
  825:             return self._state
