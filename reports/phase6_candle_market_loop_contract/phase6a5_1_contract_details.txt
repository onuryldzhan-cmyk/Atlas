    def start(self) -> bool:
        """
        Start the market worker thread.

        Returns:
            True when the loop starts or is already running.
        """

        with self.lock:
            if self.running:
                logger.warning(
                    "Market Loop is already running"
                )
                return True

            self.running = True
            self.stop_event.clear()

            self.started_at = datetime.now(
                UTC
            ).isoformat()

            self.stopped_at = None

            self.thread = threading.Thread(
                target=self.worker,
                name="AtlasMarketLoop",
                daemon=True,
            )

            self.thread.start()

        logger.info(
            "Market Loop started"
        )

        return True

METHOD stop(self) -> bool [214-266]
    def stop(self) -> bool:
        """
        Stop the worker thread gracefully.
        """

        with self.lock:
            if not self.running:
                logger.info(
                    "Market Loop is already stopped"
                )
                return True

            self.running = False
            self.stop_event.set()

            thread = self.thread

        if (
            thread is not None
            and thread.is_alive()
            and thread is not threading.current_thread()
        ):
            thread.join(
                timeout=max(
                    5.0,
                    self.interval + 1.0,
                )
            )

        with self.lock:
            still_alive = bool(
                self.thread
                and self.thread.is_alive()
            )

            self.stopped_at = datetime.now(
                UTC
            ).isoformat()

            if not still_alive:
                self.thread = None

        if still_alive:
            logger.warning(
                "Market Loop thread did not stop within timeout"
            )
            return False

        logger.info(
            "Market Loop stopped"
        )

        return True

        self,

        try:
            candles = self._load_candles(
                symbol
            )

            if not candles:
                self._increment_metric(
                    "symbols_skipped"
                )
                return None

            normalized_candles = self.processor.normalize(
                candles
            )

            if not normalized_candles:
                self._increment_metric(
                    "symbols_skipped"
                )
                return None

            if len(normalized_candles) < 50:
                logger.warning(
                    "%s skipped: insufficient candles=%s",
                    symbol,
                    len(normalized_candles),
                )

                self._increment_metric(
                    "symbols_skipped"
                )

                return None

            analysis = self.technical.analyze(
                normalized_candles
            )

            if not isinstance(
                analysis,
                dict,
            ):
                logger.warning(
                    "%s skipped: invalid technical analysis",
                    symbol,
                )

                self._increment_metric(
                    "symbols_skipped"
                )

                return None

            price = self.processor.latest_price(
                normalized_candles
            )

            price = self._safe_float(
                price
            )

            if price <= 0:
                logger.warning(
                    "%s skipped: invalid price=%s",
                    symbol,
                    price,
                )

                self._increment_metric(
                    "symbols_skipped"
                )

                return None

            score = self.calculate_score(
                analysis
            )

            market_data = self.world.build_market_data(
                symbol,
                price,
                analysis,
                score,
            )

            raw_decision = self.brain.analyze_and_decide(
                symbol,
                market_data,
            )

            self._increment_metric(
                "decisions_received"
            )

            normalized_decision = (
                self.normalize_decision(
                    raw_decision=raw_decision,
                    symbol=symbol,
                    price=price,
                    market_data=market_data,
                )
            )

            if normalized_decision is None:
                self._increment_metric(
                    "decisions_rejected"
                )

                logger.warning(
                    "%s decision normalization failed",
                    symbol,
                )

                self.publish(
                    "DECISION_NORMALIZATION_FAILED",
                    {
                        "symbol": symbol,
                        "raw_decision": self._safe_payload(
                            raw_decision
                        ),
                        "timestamp": datetime.now(
                            UTC
                        ).isoformat(),
                    },
                )

                return None

            self._increment_metric(
                "decisions_normalized"
            )

            with self.lock:
                self.last_signals[
                    symbol
                ] = normalized_decision

            self.publish(
                "ATLAS_DECISION",
                normalized_decision,
            )

            action = normalized_decision.get(
                "action",
                "HOLD",
            )

            if action in self.NON_EXECUTION_ACTIONS:
                self._increment_metric(
                    "hold_decisions"
                )

                logger.info(
                    "%s decision HOLD | execution skipped",
                    symbol,
                )

                result = {
                    "status": "SKIPPED",
                    "reason": "NON_EXECUTION_ACTION",
                    "symbol": symbol,
                    "action": action,
                    "timestamp": datetime.now(
                        UTC
                    ).isoformat(),
                }

                with self.lock:
                    self.last_executions[
                        symbol
                    ] = result

                return result

            if action not in self.VALID_EXECUTION_ACTIONS:
                self._increment_metric(
                    "decisions_rejected"
                )

                logger.warning(
                    "%s invalid normalized action: %s",
                    symbol,
                    action,
                )

                return {
                    "status": "REJECTED",
                    "reason": "INVALID_NORMALIZED_ACTION",
                    "symbol": symbol,
                    "action": action,
                }

            if self.execution is None:
                logger.warning(
                    "%s execution skipped: manager unavailable",
                    symbol,
                )

                return {
                    "status": "SKIPPED",
                    "reason": "EXECUTION_MANAGER_UNAVAILABLE",
                    "symbol": symbol,
                    "action": action,
                }

            market_state = self.world.build_market_state(
                market_data
            )

            result = self.route_execution(
                decision=normalized_decision,
                market_state=market_state,
            )

            with self.lock:
                self.last_executions[
                    symbol
                ] = result

            self._increment_metric(
                "symbols_scanned"
            )

            return result

        except Exception as exc:
            self._record_error(
                phase=f"symbol:{symbol}",
                error=exc,
            )

            logger.exception(
                "%s processing failed",
                symbol,
            )

            return None

            normalized.get(
                "action",

        normalized["symbol"] = str(

        normalized["price"] = self._safe_float(
            normalized.get(

                price,
            )
        )

        normalized["confidence"] = (
            self._normalize_confidence(
                normalized.get(
                    "confidence",
                    0,
                )
            )
        )

        normalized.setdefault(
            "strategy",
            "Atlas",
        )

        normalized.setdefault(
            "risk_level",
            "MEDIUM",
        )

        metadata = normalized.get(
            "metadata"
        )

        if not isinstance(
            metadata,
            dict,

            metadata = {}

        metadata.update(
            {
                "market_loop_version": self.VERSION,

                decision,
                market_state,
            )

        except Exception as exc:
            self._increment_metric(
                "execution_failures"
            )

            self._record_error(
                phase="execution",
                error=exc,
            )

            logger.exception(
                "%s execution routing failed",
                decision.get(
                    "symbol",
                    "UNKNOWN",
                ),
            )

            return {
                "status": "FAILED",
                "reason": "EXECUTION_EXCEPTION",
                "error": str(
                    exc
                ),
                "symbol": decision.get(
                    "symbol",
                    "UNKNOWN",
                ),
                "action": decision.get(
                    "action",
                    "UNKNOWN",
                ),
                "timestamp": datetime.now(
                    UTC
                ).isoformat(),
            }

        result_dict = self._normalize_execution_result(
            result
        )

        status = str(
            result_dict.get(
                "status",
                "UNKNOWN",
            )
        ).upper()

        if status in {
            "PAPER_EXECUTED",
            "EXECUTED",
            "COMPLETED",
            "SUCCESS",
        }:
            self._increment_metric(
                "executions_completed"
            )

        elif status == "REJECTED":
            self._increment_metric(
                "executions_rejected"
            )

        elif status in {
            "FAILED",
            "ERROR",
        }:
            self._increment_metric(
                "execution_failures"
            )

        logger.info(
            "%s execution %s | action=%s",
            decision.get(
                "symbol",
                "UNKNOWN",
            ),
            status,
            decision.get(
                "action",
                "UNKNOWN",
            ),
        )

        return result_dict

METHOD calculate_score(analysis: dict[str, Any]) -> float [1006-1061]
    def calculate_score(
        analysis: dict[str, Any],
    ) -> float:
        """
        Produce a bounded market quality score between 0 and 1.
        """

        score = 0.5

        rsi = analysis.get(
            "rsi"
        )

        trend = str(
            analysis.get(
                "trend",
                "",
            )
        ).upper()

        try:
            rsi_value = float(
                rsi
            )
        except (
            TypeError,
            ValueError,
        ):
            rsi_value = None

        if rsi_value is not None:
            if 40 <= rsi_value <= 60:
                score += 0.15

            elif rsi_value < 30:
                score += 0.10

            elif rsi_value > 70:
                score -= 0.10

        if trend == "BULLISH":
            score += 0.20

        elif trend == "BEARISH":
            score -= 0.20

        return round(
            max(
                0.0,
                min(
                    score,
                    1.0,
                ),
            ),
            2,
        )
