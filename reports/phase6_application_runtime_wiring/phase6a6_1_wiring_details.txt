METHOD execute(self) -> int [499-558]
    def execute(self) -> int:
        """
        Run the full Atlas process lifecycle.
        """

        self.install_signal_handlers()

        try:
            if not self.start():
                return self.exit_code

            self.run_forever()

        except KeyboardInterrupt:
            self.shutdown_reason = (
                "Keyboard interrupt"
            )

            logger.warning(
                "Keyboard interrupt received"
            )

        except Exception as exc:
            self.exit_code = 1

            self.shutdown_reason = (
                f"Fatal error: {exc}"
            )

            logger.critical(
                "Fatal Atlas error: %s",
                exc,
            )

            logger.debug(
                "Fatal traceback:\n%s",
                traceback.format_exc(),
            )

        finally:
            try:
                self.stop()
            except Exception:
                self.exit_code = 1

                logger.exception(
                    "Fatal error during Atlas shutdown"
                )

            logger.info(
                "Atlas process finished | "
                "exit_code=%s | reason=%s",
                self.exit_code,
                (
                    self.shutdown_reason
                    or "Normal shutdown"
                ),
            )

        return self.exit_code

    def start(self) -> bool:
        """
        Start Atlas services in dependency-safe order.

        Returns:
            True when Atlas starts successfully.
            False when startup fails.
        """

        if self.running:
            logger.warning(
                "Atlas is already running",
            )
            return True

        self.started_services.clear()
        self.stopped_at = None

        try:
            logger.info(
                "=================================================="
            )
            logger.info(
                "ATLAS STARTUP SEQUENCE"
            )
            logger.info(
                "=================================================="
            )

            self.runtime.boot()
            self.runtime.initialize()

            self.started_at = datetime.now(UTC).isoformat()

            # -------------------------------------------------
            # Execution subsystem
            # -------------------------------------------------

            self.execution.start()
            self.started_services.append("execution")

            # -------------------------------------------------
            # Exchange connection
            # -------------------------------------------------

            self.runtime.connect()

            self.market_feed.start()
            self.started_services.append("market_feed")

            # -------------------------------------------------
            # Intelligence
            # -------------------------------------------------

            self._start_optional_service(
                name="brain",
                service=self.brain,
            )

            # -------------------------------------------------
            # Runtime ready
            # -------------------------------------------------

            self.runtime.ready()

            # -------------------------------------------------
            # Market loop
            # -------------------------------------------------

            self.market_loop.start()
            self.started_services.append("market_loop")

            # -------------------------------------------------
            # Controller
            # -------------------------------------------------

            self.controller.start()
            self.started_services.append("controller")

            self.runtime.start()
            self.running = True

            logger.info(
                "ATLAS SYSTEM ONLINE"
            )
            logger.info(
                "Started services: %s",
                self.started_services,
            )

            return True

        except Exception as exc:
            self._record_error(
                phase="startup",
                error=exc,
            )

            logger.exception(
                "Atlas startup failed"
            )

            try:
                self.runtime.fail(
                    str(exc),
                )
            except Exception:
                logger.exception(
                    "Runtime failure transition failed"
                )

            self._rollback_startup()
            return False

    def stop(self) -> bool:
        """
        Stop Atlas services in reverse dependency order.
        """

        logger.info(
            "=================================================="
        )
        logger.info(
            "ATLAS SHUTDOWN SEQUENCE"
        )
        logger.info(
            "=================================================="
        )

        shutdown_errors = 0

        try:
            self.runtime.stop()
        except Exception as exc:
            shutdown_errors += 1
            self._record_error(
                phase="runtime_stop",
                error=exc,
            )

        shutdown_order = (
            ("controller", self.controller),
            ("market_loop", self.market_loop),
            ("execution", self.execution),
            ("market_feed", self.market_feed),
            ("scheduler", self.scheduler),
        )

        for name, service in shutdown_order:
            try:
                self._stop_service(
                    name=name,
                    service=service,
                )
            except Exception as exc:
                shutdown_errors += 1
                self._record_error(
                    phase=f"shutdown:{name}",
                    error=exc,
                )

                logger.exception(
                    "Service shutdown failed: %s",
                    name,
                )

        try:
            self.lifecycle.stop_all()
        except Exception as exc:
            shutdown_errors += 1
            self._record_error(
                phase="lifecycle_stop_all",
                error=exc,
            )

            logger.exception(
                "Lifecycle shutdown failed"
            )

        self.running = False
        self.stopped_at = datetime.now(UTC).isoformat()
        self.started_services.clear()

        try:
            self.runtime.shutdown()
        except Exception as exc:
            shutdown_errors += 1
            self._record_error(
                phase="runtime_shutdown",
                error=exc,
            )

        if shutdown_errors:
            logger.warning(
                "Atlas stopped with %s shutdown error(s)",
                shutdown_errors,
            )
            return False

        logger.info(
            "ATLAS SYSTEM OFFLINE"
        )
        return True

    def start(self) -> bool:
        """

        Returns:

        """

        with self.lock:
            if self.running:
                logger.warning(
                    "Market Loop is already running"

                return True

            self.running = True
            self.stop_event.clear()

            self.started_at = datetime.now(
                UTC
            ).isoformat()

            self.stopped_at = None

            self.thread = threading.Thread(
                target=self.worker,
                name="AtlasMarketLoop",
                daemon=True,
            )

            self.thread.start()

        logger.info(
            "Market Loop started"
        )

        return True

METHOD stop(self) -> bool [214-266]
    def stop(self) -> bool:
        """
        Stop the worker thread gracefully.
        """

        with self.lock:
            if not self.running:
                logger.info(
                    "Market Loop is already stopped"
                )
                return True

            self.running = False
            self.stop_event.set()

            thread = self.thread

        if (
            thread is not None
            and thread.is_alive()
            and thread is not threading.current_thread()
        ):
            thread.join(
                timeout=max(
                    5.0,
                    self.interval + 1.0,
                )
            )

        with self.lock:
            still_alive = bool(
                self.thread
                and self.thread.is_alive()
            )

            self.stopped_at = datetime.now(
                UTC
            ).isoformat()

            if not still_alive:
                self.thread = None

        if still_alive:
            logger.warning(
                "Market Loop thread did not stop within timeout"
            )
            return False

        logger.info(
            "Market Loop stopped"
        )

        return True

METHOD worker(self) -> None [272-324]
    def worker(self) -> None:
        """
        Execute continuous market cycles until shutdown.
        """

        logger.debug(
            "Market Loop worker entered"
        )

        while self.running:
            cycle_started = time.perf_counter()

            try:
                self.run_cycle()

            except Exception as exc:
                self._record_error(
                    phase="cycle",
                    error=exc,
                )

                logger.exception(
                    "Market cycle failed"
                )

            cycle_duration = (
                time.perf_counter()
                - cycle_started
            )

            with self.lock:
                self.metrics["cycles"] += 1
                self.metrics["latency"] = round(
                    cycle_duration,
                    4,
                )

                self.last_cycle_duration = round(
                    cycle_duration,
                    4,
                )

                self.last_cycle = datetime.now(
                    UTC
                ).isoformat()

            self.stop_event.wait(
                self.interval
            )

        logger.debug(
            "Market Loop worker exited"
        )
