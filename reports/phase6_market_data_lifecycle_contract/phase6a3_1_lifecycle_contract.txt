    def connect(
        self,

        """
        Validate public Binance connectivity.

        """

        with self._lock:
            if self._closed:
                raise MarketDataConnectionError(
                    "Binance public provider has already "
                    "been permanently closed."
                )

            if self._connected:
                return True

        try:
            self.get_price(
                self._connection_test_symbol
            )

        except Exception as error:
            with self._lock:
                self._connected = False

            raise MarketDataConnectionError(
                "Binance public connectivity test failed."
            ) from error

        with self._lock:
            self._connected = True

        logger.info(
            "%s connected successfully.",
            self.PROVIDER_NAME,
        )

        return True

    def get_candles(
        self,
        symbol: str,
        timeframe: str = "5m",
        limit: int = 100,
    ) -> Sequence[MarketCandle]:
        """
        Fetch and normalize Binance OHLCV candles.
        """

        normalized_symbol = (
            self._normalize_symbol(
                symbol
            )
        )

        normalized_timeframe = (
            self._normalize_timeframe(
                timeframe
            )
        )

        normalized_limit = (
            self._validate_candle_limit(
                limit
            )
        )

        payload = self._request_json(
            endpoint="/api/v3/klines",
            params={
                "symbol": normalized_symbol,
                "interval": normalized_timeframe,
                "limit": normalized_limit,
            },
        )

        if not isinstance(
            payload,
            list,
        ):
            raise MarketDataResponseError(
                "Binance kline response must be "
                "a JSON array."
            )

        candles: list[
            MarketCandle
        ] = []

        for index, raw_candle in enumerate(
            payload
        ):
            try:
                candle = self._parse_candle(
                    raw_candle,
                    symbol=normalized_symbol,
                    timeframe=(
                        normalized_timeframe
                    ),
                )

            except (
                InvalidCandleError,
                MarketDataResponseError,
                TypeError,
                ValueError,
                IndexError,
            ) as error:
                raise InvalidCandleError(
                    "Invalid Binance candle at index "
                    f"{index}: {error}"
                ) from error

            candles.append(
                candle
            )

        candles.sort(
            key=lambda item: item.open_time
        )

        with self._lock:
            self._connected = True

        return tuple(
            candles
        )

        """
        Return the latest normalized market price.

        normalized_symbol = (
            self._normalize_symbol(
                symbol
            )
        )

        requested_at = utc_now()

        with self._lock:
            self._last_request_at = requested_at

        self._publish_event(
            MarketFeedEventType.PRICE_REQUESTED,
            payload={
                "symbol": normalized_symbol,
                "provider": self.provider_name,
            },
            correlation_id=correlation_id,
        )

        started = perf_counter()

        try:
            price = self._provider.get_price(
                normalized_symbol
            )

            self._validate_price_result(
                price,
                expected_symbol=(
                    normalized_symbol
                ),
            )

        except Exception as error:
            latency = self._safe_latency(
                started
            )

            self._record_failure(
                request_type="price",
                symbol=normalized_symbol,
                latency_seconds=latency,
                error=error,
                correlation_id=correlation_id,
            )

            raise

        latency = self._safe_latency(
            started
        )

        self._record_price_success(
            price=price,
            latency_seconds=latency,
            correlation_id=correlation_id,
        )

        return price

    def start(self) -> None:
        """
        Start the service safely.

        Calling start while already running is treated as an
        idempotent operation and produces no additional side effects.
        """

        with self._lock:

            if self._state == ServiceState.RUNNING:
                return

            if self._state in {
                ServiceState.STARTING,
                ServiceState.STOPPING,
            }:
                raise ServiceLifecycleError(
                    f"Service '{self.identity.name}' cannot start "
                    f"while state is {self._state.value}."
                )

            self._state = ServiceState.STARTING
            self._last_error = None

            try:

                self._on_start()

                self._started_at = datetime.now(UTC)
                self._stopped_at = None
                self._state = ServiceState.RUNNING

                logger.info(
                    "AtlasOS service started: %s v%s",
                    self.identity.name,
                    self.identity.version,
                )

            except Exception as exc:

                self._last_error = (
                    f"{type(exc).__name__}: {exc}"
                )

                self._state = ServiceState.FAILED

                logger.exception(
                    "AtlasOS service start failed: %s",
                    self.identity.name,
                )

                raise

    def stop(self) -> None:
        """
        Stop the service safely.

        Calling stop before startup or after shutdown is treated as
        an idempotent operation.
        """

        with self._lock:

            if self._state in {
                ServiceState.CREATED,
                ServiceState.STOPPED,
            }:
                self._state = ServiceState.STOPPED

                if self._stopped_at is None:
                    self._stopped_at = datetime.now(UTC)

                return

            if self._state == ServiceState.STOPPING:
                return

            if self._state == ServiceState.STARTING:
                raise ServiceLifecycleError(
                    f"Service '{self.identity.name}' cannot stop "
                    "while startup is in progress."
                )

            self._state = ServiceState.STOPPING

            try:

                self._on_stop()

                self._stopped_at = datetime.now(UTC)
                self._state = ServiceState.STOPPED

                logger.info(
                    "AtlasOS service stopped: %s",
                    self.identity.name,
                )

            except Exception as exc:

                self._last_error = (
                    f"{type(exc).__name__}: {exc}"
                )

                self._state = ServiceState.FAILED

                logger.exception(
                    "AtlasOS service stop failed: %s",
                    self.identity.name,
                )

                raise

METHOD health(self) -> HealthReport lines 291-345
    def health(self) -> HealthReport:
        """
        Return a standardized service health report.
        """

        with self._lock:

            state = self._state

            if state == ServiceState.RUNNING:
                status = HealthStatus.HEALTHY
                message = "Service is operating normally."

            elif state == ServiceState.DEGRADED:
                status = HealthStatus.DEGRADED
                message = "Service is operational but degraded."

            elif state == ServiceState.FAILED:
                status = HealthStatus.UNHEALTHY
                message = (
                    self._last_error
                    or "Service lifecycle failed."
                )

            elif state in {
                ServiceState.CREATED,
                ServiceState.STOPPED,
            }:
                status = HealthStatus.UNKNOWN
                message = "Service is not currently running."

            else:
                status = HealthStatus.UNKNOWN
                message = (
                    f"Service lifecycle transition: "
                    f"{state.value}."
                )

            details = dict(
                self._health_details()
            )

            if self._last_error is not None:
                details.setdefault(
                    "last_error",
                    self._last_error,
                )

            return HealthReport(
                service_name=self.identity.name,
                status=status,
                state=state,
                message=message,
                details=details,
            )
