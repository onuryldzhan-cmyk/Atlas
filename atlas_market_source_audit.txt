"""
Atlas Binance Market Provider v5.0

Live market data connector.
"""


import requests
import logging



logger = logging.getLogger(
    "AtlasAI.BinanceProvider"
)



class BinanceProvider:


    VERSION = "5.0"


    BASE_URL = (
        "https://api.binance.com"
    )



    def __init__(self):

        logger.info(
            "Atlas Binance Provider v5.0 initialized"
        )




    def get_price(
        self,
        symbol
    ):


        url = (
            f"{self.BASE_URL}"
            f"/api/v3/ticker/price"
        )


        response = requests.get(

            url,

            params={
                "symbol":symbol
            },

            timeout=5

        )


        data = response.json()


        return float(
            data["price"]
        )




    def get_volume(
        self,
        symbol
    ):


        url = (
            f"{self.BASE_URL}"
            f"/api/v3/ticker/24hr"
        )


        response = requests.get(

            url,

            params={
                "symbol":symbol
            },

            timeout=5

        )


        data=response.json()


        return float(
            data["volume"]
        )





    def get_candles(
        self,
        symbol,
        timeframe="5m",
        limit=100
    ):


        url = (
            f"{self.BASE_URL}"
            f"/api/v3/klines"
        )


        response=requests.get(

            url,

            params={

                "symbol":symbol,

                "interval":timeframe,

                "limit":limit

            },

            timeout=5

        )



        candles=[]


        for item in response.json():


            candles.append({

                "open":
                float(item[1]),


                "high":
                float(item[2]),


                "low":
                float(item[3]),


                "close":
                float(item[4]),


                "volume":
                float(item[5])

            })



        return candles

"""
Atlas Candle Manager v5.1

Responsible for:
- candle storage
- normalization
- market history preparation
"""


import logging
from datetime import datetime, UTC



logger = logging.getLogger(
    "AtlasAI.CandleManager"
)



class CandleManager:


    VERSION = "5.1"



    def __init__(
        self,
        provider=None
    ):

        self.provider = provider

        self.cache = {}

        logger.info(
            "Atlas Candle Manager v5.1 initialized"
        )



    # =====================================
    # GET MARKET CANDLES
    # =====================================


    def get_candles(
        self,
        symbol,
        timeframe="5m",
        limit=100
    ):


        if not self.provider:

            return []



        candles = self.provider.get_candles(

            symbol,

            timeframe,

            limit

        )


        self.cache[symbol] = {

            "candles": candles,

            "time":
            datetime.now(
                UTC
            ).isoformat()

        }



        return candles





    # =====================================
    # LAST PRICE
    # =====================================


    def last_price(
        self,
        symbol
    ):


        candles = self.cache.get(
            symbol
        )


        if not candles:

            return None



        data = candles["candles"]


        if not data:

            return None



        return data[-1]["close"]






    # =====================================
    # MARKET SNAPSHOT
    # =====================================


    def snapshot(
        self,
        symbol
    ):


        candles = self.get_candles(
            symbol
        )


        if not candles:

            return {

                "symbol":
                symbol,

                "status":
                "NO_DATA"

            }



        last = candles[-1]



        return {

            "symbol":
            symbol,


            "price":
            last["close"],


            "high":
            last["high"],


            "low":
            last["low"],


            "volume":
            last["volume"],


            "candles":
            len(candles),


            "time":
            datetime.now(
                UTC
            ).isoformat()

        }






    # =====================================
    # STATUS
    # =====================================


    def status(self):


        return {

            "version":
            self.VERSION,


            "cached_symbols":
            list(
                self.cache.keys()
            )

        }

"""
Atlas Candle Processor v2.0

Exchange candle normalization layer.

Responsibilities:

- Convert Binance candles
- Validate OHLCV
- Remove invalid candles
- Sort candles
- Prepare data for intelligence layer

"""

from __future__ import annotations

import logging
from datetime import datetime, timezone


logger = logging.getLogger(
    "AtlasAI.CandleProcessor"
)


class CandleProcessor:


    def __init__(self):

        self.version = "2.0"



    def normalize(
        self,
        candles
    ):

        if not candles:

            return []


        result = []


        for candle in candles:


            try:


                normalized = self._parse(
                    candle
                )


                if normalized:

                    result.append(
                        normalized
                    )


            except Exception as e:

                logger.warning(
                    f"Candle skipped: {e}"
                )



        result = self._clean(
            result
        )


        return result



    # =================================
    # PARSER
    # =================================


    def _parse(
        self,
        candle
    ):


        if isinstance(
            candle,
            list
        ):


            return {

                "timestamp":
                    datetime.now(
                        timezone.utc
                    ).isoformat(),


                "open":
                    float(candle[1]),


                "high":
                    float(candle[2]),


                "low":
                    float(candle[3]),


                "close":
                    float(candle[4]),


                "volume":
                    float(candle[5])

            }



        elif isinstance(
            candle,
            dict
        ):


            return {

                "timestamp":
                    candle.get(
                        "timestamp",
                        datetime.now(
                            timezone.utc
                        ).isoformat()
                    ),


                "open":
                    float(
                        candle["open"]
                    ),


                "high":
                    float(
                        candle["high"]
                    ),


                "low":
                    float(
                        candle["low"]
                    ),


                "close":
                    float(
                        candle["close"]
                    ),


                "volume":
                    float(
                        candle.get(
                            "volume",
                            0
                        )
                    )

            }


        return None



    # =================================
    # VALIDATION
    # =================================


    def _clean(
        self,
        candles
    ):


        cleaned = []


        for c in candles:


            if (

                c["high"] < c["low"]

                or

                c["close"] <= 0

                or

                c["open"] <= 0

            ):

                continue



            cleaned.append(
                c
            )



        return cleaned



    # =================================
    # HELPERS
    # =================================


    def latest_price(
        self,
        candles
    ):


        if not candles:

            return None


        return candles[-1]["close"]



    def candle_count(
        self,
        candles
    ):


        return len(
            candles
        )

"""
Atlas Binance Global Adapter v2.0

Production exchange connector.

Features:

- Safe connection handling
- Automatic reconnect
- Connection validation
- Paper trading protection
- Binance API wrapper

"""


from __future__ import annotations


import logging
import os
import time


from dotenv import load_dotenv
from binance.spot import Spot


from exchange.base_exchange import BaseExchange



load_dotenv()



logger = logging.getLogger(
    "AtlasAI.BinanceGlobal"
)





class BinanceGlobalExchange(BaseExchange):


    VERSION = "2.0"



    def __init__(self):


        self.connected = False


        self.client = None


        self.api_key = os.getenv(
            "BINANCE_API_KEY",
            ""
        )


        self.api_secret = os.getenv(
            "BINANCE_API_SECRET",
            ""
        )



        logger.info(
            "Binance Global Adapter initialized"
        )






    # =====================================
    # CONNECT
    # =====================================


    def connect(self):


        try:


            if self.client is None:


                self.client = Spot(

                    api_key=self.api_key,

                    api_secret=self.api_secret

                )



            # connection test


            self.client.time()



            self.connected = True



            logger.info(
                "Binance Global connected"
            )



            return True



        except Exception as e:


            self.connected = False


            logger.exception(
                f"Binance connection failed: {e}"
            )


            return False







    # =====================================
    # DISCONNECT
    # =====================================


    def disconnect(self):


        self.connected = False


        self.client = None



        logger.info(
            "Binance disconnected"
        )






    # =====================================
    # ENSURE CONNECTION
    # =====================================


    def _ensure_connection(self):


        if not self.connected or self.client is None:


            logger.warning(
                "Binance reconnect required"
            )


            self.connect()



        if self.client is None:


            raise ConnectionError(
                "Binance client unavailable"
            )








    # =====================================
    # PRICE
    # =====================================


    def get_price(
        self,
        symbol
    ):


        self._ensure_connection()



        ticker = self.client.ticker_price(

            symbol=symbol

        )



        return float(
            ticker["price"]
        )








    # =====================================
    # CANDLES
    # =====================================


    def get_klines(

        self,

        symbol,

        interval="1m",

        limit=500

    ):


        self._ensure_connection()



        return self.client.klines(

            symbol=symbol,

            interval=interval,

            limit=limit

        )








    # =====================================
    # ORDER
    # =====================================


    def place_order(

        self,

        symbol,

        side,

        quantity

    ):


        logger.warning(
            "Real order disabled - PAPER MODE"
        )



        return {


            "status":
            "PAPER_ONLY",


            "symbol":
            symbol,


            "side":
            side,


            "quantity":
            quantity


        }
